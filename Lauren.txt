#
# The function is expected to return an INTEGER.

def minimumLoss(price):
    # Store price with original index
    price_with_index = [(p, i) for i, p in enumerate(price)]
    
    # Sort by price
    price_with_index.sort()
    
    min_loss = float('inf')
    
    # Check adjacent pairs
    for i in range(1, len(price_with_index)):
        lower_price, lower_idx = price_with_index[i-1]
        higher_price, higher_idx = price_with_index[i]
        
        # Ensure buying happens before selling
        if higher_idx < lower_idx:
            loss = higher_price - lower_price
            min_loss = min(min_loss, loss)
    
    return min_loss

if __name__ == '__main__':
    fptr = open(os.environ['OUTPUT_PATH'], 'w')

    price = list(map(int, input().rstrip().split()))

    result = minimumLoss(price)
